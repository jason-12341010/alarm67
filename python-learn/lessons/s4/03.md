**裝飾器（decorator）** 可以在**不修改原函式程式碼**的情況下，幫函式加上額外的功能：計時、記錄、快取、權限檢查… 你在 Flask（`@app.route`）、FastAPI、pytest 都會大量看到它。

## 裝飾器就是「接收函式、回傳新函式」的函式

```python
def shout(func):
    def wrapper():
        result = func()
        return result.upper() + "!!!"
    return wrapper

def hello():
    return "hello"

hello = shout(hello)        # 用 shout 包裝 hello
print(hello())
```

## @ 語法糖

`@shout` 寫在函式上面，**等同於** `hello = shout(hello)`：

```python
def shout(func):
    def wrapper():
        return func().upper() + "!!!"
    return wrapper

@shout
def hello():
    return "hello"

@shout
def bye():
    return "bye"

print(hello(), bye())
```

## 通用的裝飾器：*args 和 **kwargs

為了讓裝飾器可以用在**任何參數**的函式上：

```python
import time

def timer(func):
    def wrapper(*args, **kwargs):
        start = time.perf_counter()
        result = func(*args, **kwargs)      # 原封不動地傳遞參數和回傳值
        ms = (time.perf_counter() - start) * 1000
        print(f"⏱ {func.__name__} 花了 {ms:.2f} ms")
        return result
    return wrapper

@timer
def total(n):
    return sum(range(n))

@timer
def greet(name, punctuation="!"):
    return f"Hi {name}{punctuation}"

print(total(1_000_000))
print(greet("小明", punctuation="～"))
```

## functools.wraps：保留原函式的資訊

包裝之後，函式的名字和說明會變成 `wrapper` 的。用 `@wraps` 修正：

```python
from functools import wraps

def bad(func):
    def wrapper(*args, **kwargs):
        return func(*args, **kwargs)
    return wrapper

def good(func):
    @wraps(func)
    def wrapper(*args, **kwargs):
        return func(*args, **kwargs)
    return wrapper

@bad
def f1():
    """我是 f1"""

@good
def f2():
    """我是 f2"""

print(f1.__name__, f1.__doc__)
print(f2.__name__, f2.__doc__)
```

> 💡 寫裝飾器時，**一律加上 `@wraps(func)`**，這是好習慣。

## 帶參數的裝飾器

`@retry(times=3)` 這種寫法要**多包一層**：最外層接收參數，回傳真正的裝飾器：

```python
from functools import wraps
import random

def retry(times=3):
    def decorator(func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            for attempt in range(1, times + 1):
                try:
                    return func(*args, **kwargs)
                except ConnectionError as e:
                    print(f"  第 {attempt} 次失敗：{e}")
            raise ConnectionError(f"{func.__name__} 重試 {times} 次都失敗")
        return wrapper
    return decorator

random.seed(11)

@retry(times=4)
def fetch_data():
    if random.random() < 0.7:
        raise ConnectionError("網路不穩")
    return "📦 資料下載成功"

print(fetch_data())
```

## 疊加多個裝飾器

由**下往上**套用（最靠近函式的先套）：

```python
from functools import wraps

def bold(func):
    @wraps(func)
    def wrapper(*a, **k):
        return f"<b>{func(*a, **k)}</b>"
    return wrapper

def italic(func):
    @wraps(func)
    def wrapper(*a, **k):
        return f"<i>{func(*a, **k)}</i>"
    return wrapper

@bold
@italic                # 等同於 text = bold(italic(text))
def text(s):
    return s

print(text("Python"))
```

## 內建的好用裝飾器

```python
from functools import lru_cache, cache
import time

@lru_cache(maxsize=None)          # 自動快取結果，相同參數直接回傳
def fib(n):
    return n if n < 2 else fib(n - 1) + fib(n - 2)

t = time.perf_counter()
print(fib(100))
print(f"{(time.perf_counter() - t) * 1000:.2f} ms")
print(fib.cache_info())
```

沒有快取的話，`fib(100)` 要算到天荒地老。你已經看過的 `@property`、`@classmethod`、`@staticmethod`、`@dataclass` 也都是裝飾器！

## 類別也可以當裝飾器、也可以被裝飾

```python
registry = {}

def register(cls):              # 裝飾類別：把類別登記起來
    registry[cls.__name__.lower()] = cls
    return cls

@register
class Cat:
    sound = "喵"

@register
class Dog:
    sound = "汪"

print(registry)
print(registry["dog"]().sound)
```

這就是很多框架「**外掛 / 登記**」機制的原理，例如 Flask 用 `@app.route("/")` 把函式登記成網址的處理函式。

:::exercise 記錄函式呼叫
寫裝飾器 `log_calls`：每次呼叫被裝飾的函式時，**印出**一行 `呼叫 函式名(參數)` ，然後回傳原本的結果。參數格式：
- 位置引數用 `repr()`，例如 `add(1, 2)`、`greet('Amy')`
- 關鍵字引數寫成 `key=repr(value)`，例如 `greet('Amy', punct='!')`

記得使用 `@wraps`。
@@starter
from functools import wraps


def log_calls(func):
    pass


@log_calls
def add(a, b):
    return a + b


@log_calls
def greet(name, punct="."):
    return f"Hi {name}{punct}"


print(add(1, 2))
print(greet("Amy", punct="!"))
@@check
assert lines == ["呼叫 add(1, 2)", "3", "呼叫 greet('Amy', punct='!')", "Hi Amy!"], f"輸出不對：{lines}"
add = ns["add"]
assert add.__name__ == "add", "請使用 @wraps(func) 保留函式名稱"
@ns["log_calls"]
def none():
    return 42
import io, contextlib
buf = io.StringIO()
with contextlib.redirect_stdout(buf):
    r = none()
assert r == 42 and buf.getvalue().strip() == "呼叫 none()", "沒有參數時要印出 none()"
@@hint
`parts = [repr(a) for a in args] + [f"{k}={v!r}" for k, v in kwargs.items()]`，然後 `print(f"呼叫 {func.__name__}({', '.join(parts)})")`。
@@solution
from functools import wraps


def log_calls(func):
    @wraps(func)
    def wrapper(*args, **kwargs):
        parts = [repr(a) for a in args] + [f"{k}={v!r}" for k, v in kwargs.items()]
        print(f"呼叫 {func.__name__}({', '.join(parts)})")
        return func(*args, **kwargs)
    return wrapper


@log_calls
def add(a, b):
    return a + b


@log_calls
def greet(name, punct="."):
    return f"Hi {name}{punct}"


print(add(1, 2))
print(greet("Amy", punct="!"))

:::exercise 參數檢查裝飾器
寫帶參數的裝飾器 `require_positive(*names)`：檢查指定名稱的**關鍵字引數或位置引數**是否都 > 0，不是的話 `raise ValueError(f"{name} 必須大於 0")`。

提示：用 `inspect.signature(func).bind(*args, **kwargs)` 可以把所有引數對應到參數名稱。
@@starter
import inspect
from functools import wraps


def require_positive(*names):
    pass


@require_positive("price", "qty")
def subtotal(price, qty, discount=0):
    return price * qty - discount


print(subtotal(100, 3))
print(subtotal(qty=2, price=50, discount=10))
try:
    subtotal(100, 0)
except ValueError as e:
    print("❌", e)
@@check
f = ns["subtotal"]
assert f(100, 3) == 300 and f(qty=2, price=50, discount=10) == 90
assert f.__name__ == "subtotal", "請使用 @wraps"
for args, kwargs, bad in [((100, 0), {}, "qty"), ((), {"price": -1, "qty": 1}, "price"), ((5,), {"qty": -2}, "qty")]:
    try:
        f(*args, **kwargs)
    except ValueError as e:
        assert str(e) == f"{bad} 必須大於 0", f"錯誤訊息不對：{e}"
    else:
        raise AssertionError(f"subtotal(*{args}, **{kwargs}) 應該 raise ValueError")
assert f(1, 1, discount=-5) == 6, "沒有指定檢查的參數（discount）不用檢查"
@@hint
三層：`def require_positive(*names): def decorator(func): @wraps(func) def wrapper(*args, **kwargs): ...`。在 wrapper 裡用 `bound = inspect.signature(func).bind(*args, **kwargs)`，然後檢查 `bound.arguments[name]`。
@@solution
import inspect
from functools import wraps


def require_positive(*names):
    def decorator(func):
        sig = inspect.signature(func)

        @wraps(func)
        def wrapper(*args, **kwargs):
            bound = sig.bind(*args, **kwargs)
            for name in names:
                if name in bound.arguments and not bound.arguments[name] > 0:
                    raise ValueError(f"{name} 必須大於 0")
            return func(*args, **kwargs)

        return wrapper

    return decorator


@require_positive("price", "qty")
def subtotal(price, qty, discount=0):
    return price * qty - discount


print(subtotal(100, 3))
print(subtotal(qty=2, price=50, discount=10))
try:
    subtotal(100, 0)
except ValueError as e:
    print("❌", e)
