`itertools` 和 `functools` 是處理**迭代**和**函式**的兩大工具箱。熟悉它們，可以把很多迴圈縮成一行，而且通常更快。

## itertools：無限迭代器

```python
from itertools import count, cycle, repeat, islice

print(list(islice(count(10, 5), 5)))              # 10, 15, 20, …
print(list(islice(cycle("紅綠黃"), 7)))            # 無限循環
print(list(repeat("哈", 3)))

for i, player in zip(range(5), cycle(["小明", "小美", "阿華"])):
    print(f"第 {i + 1} 輪：{player}")
```

## 排列與組合

```python
from itertools import permutations, combinations, combinations_with_replacement, product

print(list(permutations("ABC", 2)))         # 排列：順序有差
print(list(combinations("ABCD", 2)))        # 組合：順序沒差
import math
print(math.comb(49, 6), "種樂透組合")          # 只要數量時用 math.comb，不要真的列出來
print(list(product("AB", [1, 2])))          # 笛卡兒積：所有配對
print(list(product([0, 1], repeat=3)))      # 3 位元的所有可能

# 實際應用：找出加總剛好 100 的兩個商品
prices = {"耳機": 45, "滑鼠": 30, "鍵盤": 55, "貼紙": 70, "線材": 25}
for (a, pa), (b, pb) in combinations(prices.items(), 2):
    if pa + pb == 100:
        print(a, "+", b)
```

## 串接、切片、分組

```python
from itertools import chain, islice, groupby, accumulate, pairwise, zip_longest

print(list(chain([1, 2], "ab", range(3))))           # 串接多個可迭代物件
print(list(chain.from_iterable([[1, 2], [3], [4, 5]])))  # 攤平一層

print(list(islice(range(100), 10, 20, 3)))           # 對任何迭代器切片

print(list(accumulate([1, 2, 3, 4, 5])))             # 累加
print(list(accumulate([3, 1, 4, 1, 5], max)))         # 累積最大值

print(list(pairwise([1, 4, 9, 16])))                 # 相鄰兩兩配對
print([b - a for a, b in pairwise([100, 120, 90, 130])])   # 每天的變化

print(list(zip_longest("abc", [1], fillvalue="-")))   # 長度不同時補值
```

### groupby：分組（要先排序！）

```python
from itertools import groupby

students = [("小明", "A"), ("小美", "B"), ("阿華", "A"), ("小強", "C"), ("小芳", "B")]
students.sort(key=lambda s: s[1])                 # groupby 只會把「相鄰」的歸在一組
for grade, group in groupby(students, key=lambda s: s[1]):
    print(grade, [name for name, _ in group])

# 壓縮連續重複的字元
s = "aaabccddddde"
print("".join(f"{ch}{len(list(g))}" for ch, g in groupby(s)))
```

## itertools：篩選

```python
from itertools import takewhile, dropwhile, filterfalse, compress

temps = [18, 20, 23, 27, 31, 29, 24]
print(list(takewhile(lambda t: t < 25, temps)))   # 一直取，直到條件不成立
print(list(dropwhile(lambda t: t < 25, temps)))   # 一直丟，直到條件不成立
print(list(filterfalse(lambda t: t % 2, temps)))  # 留下條件為 False 的
print(list(compress("ABCDE", [1, 0, 1, 0, 1])))   # 用遮罩篩選
```

## functools：函式工具

```python
from functools import reduce, partial, lru_cache, cache, cached_property, total_ordering

# reduce：累積成一個值
print(reduce(lambda a, b: a * b, range(1, 6)))

# partial：固定部分參數，產生新函式
int_from_binary = partial(int, base=2)
print(int_from_binary("1011"))

def power(base, exp):
    return base ** exp
square = partial(power, exp=2)
cube = partial(power, exp=3)
print(square(5), cube(2))
```

```python
from functools import cached_property
import time

class Report:
    def __init__(self, data):
        self.data = data

    @cached_property                  # 第一次存取時計算，之後直接用結果
    def stats(self):
        print("  （計算中…）")
        time.sleep(0.2)
        return {"sum": sum(self.data), "max": max(self.data)}

r = Report([3, 1, 4, 1, 5])
print(r.stats)
print(r.stats)                        # 不會再計算
```

### singledispatch：依照型別選擇實作

```python
from functools import singledispatch

@singledispatch
def describe(x):
    return f"不知道怎麼描述 {type(x).__name__}"

@describe.register
def _(x: int):
    return f"整數 {x}，{'偶數' if x % 2 == 0 else '奇數'}"

@describe.register
def _(x: list):
    return f"有 {len(x)} 個元素的串列"

@describe.register
def _(x: str):
    return f"長度 {len(x)} 的字串"

for v in [42, [1, 2], "hello", 3.14]:
    print(describe(v))
```

## operator：把運算子變成函式

```python
from operator import itemgetter, attrgetter, add, mul
from functools import reduce

rows = [("小明", 85, 18), ("小美", 92, 17), ("阿華", 85, 16)]
print(sorted(rows, key=itemgetter(1, 2)))      # 比 lambda r: (r[1], r[2]) 更快更清楚
print(reduce(mul, [1, 2, 3, 4]))
print(list(map(add, [1, 2], [10, 20])))
```

prices = [100, 102, 105, 103, 104, 108, 110, 111, 109, 112]

changes = [b - a for a, b in pairwise(prices)]
