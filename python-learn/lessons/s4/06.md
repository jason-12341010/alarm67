`collections` 模組提供了比 list、dict 更專門的容器。用對工具，程式可以短很多、快很多。

## Counter：計數

```python
from collections import Counter

votes = ["小明", "小美", "小明", "阿華", "小明", "小美"]
c = Counter(votes)
print(c)
print(c["小明"], c["沒人"])          # 不存在的鍵回傳 0，不會出錯
print(c.most_common(2))

c.update(["阿華", "阿華"])           # 加票
print(c)

text = "mississippi"
print(Counter(text).most_common(3))

a = Counter("aabbbc")
b = Counter("abccdd")
print(a + b, a - b, a & b, sep="\n")  # 計數也可以做運算
print(sum(a.values()), a.total())
```

## defaultdict：有預設值的字典

不存在的鍵會**自動用預設值建立**，不用再寫 `if key not in d`：

```python
from collections import defaultdict

words = ["apple", "avocado", "banana", "blueberry", "cherry", "apricot"]

groups = defaultdict(list)             # 預設值是空串列
for w in words:
    groups[w[0]].append(w)
print(dict(groups))

count = defaultdict(int)               # 預設值是 0
for w in words:
    count[len(w)] += 1
print(dict(count))

nested = defaultdict(lambda: defaultdict(int))   # 二層字典
sales = [("台北", "咖啡", 3), ("台北", "茶", 2), ("台中", "咖啡", 5), ("台北", "咖啡", 1)]
for city, item, qty in sales:
    nested[city][item] += qty
print({k: dict(v) for k, v in nested.items()})
```

## deque：雙向佇列

在**頭尾**加入、刪除都很快（list 在開頭插入很慢）：

```python
from collections import deque

q = deque([1, 2, 3])
q.append(4)            # 右邊加入
q.appendleft(0)        # 左邊加入
print(q)
print(q.pop(), q.popleft(), q)
q.rotate(1)            # 旋轉
print(q)

recent = deque(maxlen=3)          # 固定長度：滿了會自動丟掉最舊的
for page in ["首頁", "商品", "購物車", "結帳", "完成"]:
    recent.append(page)
    print(list(recent))
```

```python
import time
from collections import deque

n = 50_000
t = time.perf_counter()
lst = []
for i in range(n):
    lst.insert(0, i)
print(f"list.insert(0)：{(time.perf_counter() - t) * 1000:.0f} ms")

t = time.perf_counter()
dq = deque()
for i in range(n):
    dq.appendleft(i)
print(f"deque.appendleft：{(time.perf_counter() - t) * 1000:.0f} ms")
```

## namedtuple：有名字的 tuple

```python
from collections import namedtuple

Point = namedtuple("Point", ["x", "y"])
p = Point(3, 4)
print(p, p.x, p[1])               # 可以用名字，也可以用索引
x, y = p                          # 可以解包
print(p._replace(x=10), p._asdict())

Color = namedtuple("Color", "r g b", defaults=[0])
print(Color(255, 128))
```

> 💡 新的程式碼比較常用 `dataclass`（可以加方法、可以修改），或 `typing.NamedTuple`（有型別提示的 namedtuple）。

```python
from typing import NamedTuple

class Employee(NamedTuple):
    name: str
    dept: str
    salary: int = 40000

e = Employee("小明", "工程部")
print(e, e.salary)
```

## OrderedDict 與 ChainMap

```python
from collections import OrderedDict, ChainMap

# OrderedDict：現在一般 dict 也會保留順序，但它多了 move_to_end
od = OrderedDict(a=1, b=2, c=3)
od.move_to_end("a")
print(list(od))

# ChainMap：把多個字典串起來查詢，常用在「設定的優先順序」
defaults = {"theme": "light", "lang": "en", "font": 14}
user = {"theme": "dark"}
cli = {"font": 18}
config = ChainMap(cli, user, defaults)     # 依序查找
print(config["theme"], config["lang"], config["font"])
```

## heapq：優先佇列（雖然不在 collections，但很常一起用）

```python
import heapq

nums = [5, 1, 8, 3, 9, 2]
print(heapq.nsmallest(3, nums), heapq.nlargest(2, nums))

tasks = []
heapq.heappush(tasks, (2, "寫報告"))
heapq.heappush(tasks, (1, "修 bug"))
heapq.heappush(tasks, (3, "開會"))
while tasks:
    priority, task = heapq.heappop(tasks)   # 永遠拿出最小的
    print(priority, task)
```

from collections import Counter, defaultdict, deque

logs = [
    ("amy", "/"), ("bob", "/products"), ("amy", "/products"), ("cat", "/"),
    ("amy", "/cart"), ("bob", "/"), ("cat", "/products"), ("bob", "/products"),
    ("amy", "/checkout"), ("cat", "/about"),
]

top_pages = Counter(page for _, page in logs).most_common(3)
user_pages = defaultdict(list)
last3 = deque(maxlen=3)
for user, page in logs:
    user_pages[user].append(page)
    last3.append(page)

print(top_pages, dict(user_pages), list(last3))
