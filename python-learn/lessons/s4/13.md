第 4 階段的期末專案：用 **async、產生器、裝飾器、上下文管理器、dataclass、型別提示、正規表示式** 做一個**非同步爬蟲模擬器**。

因為在瀏覽器裡不能真的連到其他網站，我們用一個「假的網路」模擬網頁和延遲。架構和真正的爬蟲完全一樣，之後只要把 `fake_fetch` 換成 `httpx`，就能爬真的網站（第 6 階段會教）。

## 架構

```
起始網址
   ↓
[佇列 Queue] ←───────── 新找到的連結
   ↓                         ↑
多個 worker 同時抓網頁 ──→ 用 regex 解析連結和標題
   ↓
結果（dataclass）→ 統計報告
```

## 完整程式

```python
import asyncio
import random
import re
import time
from collections import Counter
from contextlib import contextmanager
from dataclasses import dataclass, field
from functools import wraps

random.seed(42)

# ---------- 假的網路：每個網址對應一段 HTML ----------
SITE = {
    "/": '<title>首頁</title><a href="/news">新聞</a><a href="/shop">商店</a><a href="/about">關於</a>',
    "/news": '<title>新聞</title><a href="/news/1">AI 新突破</a><a href="/news/2">Python 3.14</a><a href="/">首頁</a>',
    "/news/1": '<title>AI 新突破</title><a href="/news">回列表</a>',
    "/news/2": '<title>Python 3.14 發布</title><a href="/news">回列表</a><a href="/broken">壞掉的連結</a>',
    "/shop": '<title>商店</title><a href="/shop/item/1">鍵盤</a><a href="/shop/item/2">滑鼠</a>',
    "/shop/item/1": '<title>機械鍵盤</title><a href="/shop">回商店</a>',
    "/shop/item/2": '<title>無線滑鼠</title><a href="/shop">回商店</a><a href="/shop/item/1">看鍵盤</a>',
    "/about": '<title>關於我們</title>',
}


class FetchError(Exception):
    pass


async def fake_fetch(url: str) -> str:
    await asyncio.sleep(random.uniform(0.05, 0.25))      # 模擬網路延遲
    if random.random() < 0.15:
        raise FetchError(f"連線逾時：{url}")               # 模擬不穩定的網路
    if url not in SITE:
        raise FetchError(f"404 Not Found：{url}")
    return SITE[url]


# ---------- 工具：重試裝飾器、計時器 ----------
def retry(times: int = 3):
    def decorator(func):
        @wraps(func)
        async def wrapper(*args, **kwargs):
            for attempt in range(1, times + 1):
                try:
                    return await func(*args, **kwargs)
                except FetchError as e:
                    if "404" in str(e) or attempt == times:
                        raise
                    await asyncio.sleep(0.05 * attempt)       # 等一下再重試
        return wrapper
    return decorator


@contextmanager
def timer(label: str):
    t = time.perf_counter()
    yield
    print(f"⏱ {label}：{time.perf_counter() - t:.2f} 秒")


# ---------- 資料結構 ----------
@dataclass
class Page:
    url: str
    title: str
    links: list[str] = field(default_factory=list)
    depth: int = 0


@dataclass
class Crawler:
    start: str
    max_depth: int = 2
    concurrency: int = 3
    pages: dict[str, Page] = field(default_factory=dict)
    errors: dict[str, str] = field(default_factory=dict)

    TITLE = re.compile(r"<title>(.*?)</title>")
    LINK = re.compile(r'<a href="(.*?)">')

    @retry(times=3)
    async def fetch(self, url: str) -> str:
        return await fake_fetch(url)

    def parse(self, url: str, html: str, depth: int) -> Page:
        title = m.group(1) if (m := self.TITLE.search(html)) else "（沒有標題）"
        return Page(url, title, self.LINK.findall(html), depth)

    async def worker(self, name: str, queue: asyncio.Queue, seen: set[str]):
        while True:
            url, depth = await queue.get()
            try:
                html = await self.fetch(url)
                page = self.parse(url, html, depth)
                self.pages[url] = page
                print(f"  [{name}] ✓ {url:<15} {page.title}")
                if depth < self.max_depth:
                    for link in page.links:
                        if link not in seen:
                            seen.add(link)
                            queue.put_nowait((link, depth + 1))
            except FetchError as e:
                self.errors[url] = str(e)
                print(f"  [{name}] ✗ {e}")
            finally:
                queue.task_done()

    async def run(self):
        queue: asyncio.Queue = asyncio.Queue()
        seen = {self.start}
        queue.put_nowait((self.start, 0))
        workers = [asyncio.create_task(self.worker(f"w{i}", queue, seen)) for i in range(self.concurrency)]
        await queue.join()                       # 等到佇列裡的工作全部完成
        for w in workers:
            w.cancel()

    def report(self):
        print(f"\n📊 成功 {len(self.pages)} 頁，失敗 {len(self.errors)} 頁")
        by_depth = Counter(p.depth for p in self.pages.values())
        for d in sorted(by_depth):
            print(f"  深度 {d}：{by_depth[d]} 頁")
        popular = Counter(link for p in self.pages.values() for link in p.links)
        print("  最多頁面連到：", popular.most_common(3))


crawler = Crawler("/", max_depth=3, concurrency=3)
with timer("爬取完成"):
    asyncio.run(crawler.run())
crawler.report()
```

試著修改 `concurrency`（同時幾個 worker）和 `max_depth`，觀察總時間的變化。

## 這個專案用到了什麼？

| 概念 | 在哪裡 |
|---|---|
| async / await、Queue、create_task | 多個 worker 同時抓網頁 |
| 裝飾器（async 版） | `@retry` 自動重試不穩定的連線 |
| 上下文管理器 | `timer` 計時 |
| dataclass + 型別提示 | `Page`、`Crawler` |
| 正規表示式 | 解析標題和連結 |
| 海象運算子 `:=` | `if (m := ...)` 一邊賦值一邊判斷 |
| Counter | 統計報告 |
