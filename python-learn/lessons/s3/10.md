## 設計：有哪些類別？

```
Character（角色，抽象類別）
├── Hero（玩家）：有背包、會升級
└── Monster（怪物）
    ├── Slime（史萊姆）
    └── Dragon（龍）：有特殊技能

Item（道具，dataclass）
Battle（戰鬥流程）
```

## 完整程式

讀過一遍後執行看看。因為設定了 `random.seed`，每次結果都一樣；改掉 seed 的數字就會有不同的戰鬥過程。

```python
import random
from abc import ABC, abstractmethod
from dataclasses import dataclass

random.seed(7)


@dataclass
class Item:
    name: str
    heal: int = 0
    attack_bonus: int = 0


class Character(ABC):
    def __init__(self, name, hp, attack, defense):
        self.name = name
        self.max_hp = hp
        self._hp = hp
        self.attack = attack
        self.defense = defense

    @property
    def hp(self):
        return self._hp

    @hp.setter
    def hp(self, value):                     # HP 永遠在 0 ～ max_hp 之間
        self._hp = max(0, min(value, self.max_hp))

    @property
    def alive(self):
        return self.hp > 0

    def take_damage(self, raw):
        dmg = max(1, raw - self.defense)
        self.hp -= dmg
        return dmg

    def hp_bar(self, width=12):
        filled = round(width * self.hp / self.max_hp)
        return "█" * filled + "░" * (width - filled)

    def __str__(self):
        return f"{self.name:<4} {self.hp_bar()} {self.hp:>3}/{self.max_hp}"

    @abstractmethod
    def act(self, target): ...


class Hero(Character):
    def __init__(self, name):
        super().__init__(name, hp=60, attack=12, defense=3)
        self.level = 1
        self.exp = 0
        self.bag = [Item("藥水", heal=25), Item("藥水", heal=25), Item("力量藥劑", attack_bonus=5)]

    def act(self, target):
        if self.hp < self.max_hp * 0.4 and any(i.heal for i in self.bag):
            return self.use(next(i for i in self.bag if i.heal))
        if random.random() < 0.2:
            dmg = target.take_damage(self.attack * 2)
            return f"💥 {self.name} 使出會心一擊！造成 {dmg} 點傷害"
        dmg = target.take_damage(self.attack + random.randint(-2, 3))
        return f"⚔️  {self.name} 攻擊 {target.name}，造成 {dmg} 點傷害"

    def use(self, item):
        self.bag.remove(item)
        self.hp += item.heal
        self.attack += item.attack_bonus
        return f"🧪 {self.name} 使用了 {item.name}"

    def gain_exp(self, amount):
        self.exp += amount
        msgs = [f"✨ 獲得 {amount} 經驗值"]
        while self.exp >= self.level * 20:
            self.exp -= self.level * 20
            self.level += 1
            self.max_hp += 15
            self.hp = self.max_hp
            self.attack += 3
            msgs.append(f"🎉 升級到 Lv.{self.level}！HP 全滿，攻擊力 {self.attack}")
        return msgs


class Monster(Character):
    exp_reward = 10

    def act(self, target):
        dmg = target.take_damage(self.attack + random.randint(-1, 2))
        return f"👊 {self.name} 攻擊，{target.name} 受到 {dmg} 點傷害"


class Slime(Monster):
    exp_reward = 12

    def __init__(self):
        super().__init__("史萊姆", hp=30, attack=7, defense=1)


class Dragon(Monster):
    exp_reward = 100

    def __init__(self):
        super().__init__("巨龍", hp=110, attack=14, defense=4)

    def act(self, target):                        # 多型：龍有自己的攻擊方式
        if random.random() < 0.3:
            dmg = target.take_damage(22)
            return f"🔥 {self.name} 噴火！{target.name} 受到 {dmg} 點傷害"
        return super().act(target)


class Battle:
    def __init__(self, hero, monster):
        self.hero = hero
        self.monster = monster
        self.round = 0

    def run(self):
        print(f"\n===== ⚠️  {self.monster.name} 出現了！ =====")
        while self.hero.alive and self.monster.alive:
            self.round += 1
            print(f"-- 第 {self.round} 回合 --")
            for attacker, defender in [(self.hero, self.monster), (self.monster, self.hero)]:
                if attacker.alive:
                    print(attacker.act(defender))
            print(self.hero)
            print(self.monster)
        if self.hero.alive:
            print(f"🏆 打倒了 {self.monster.name}！")
            for msg in self.hero.gain_exp(self.monster.exp_reward):
                print(msg)
            return True
        print(f"💀 {self.hero.name} 倒下了…")
        return False


hero = Hero("勇者")
for monster in [Slime(), Slime(), Dragon()]:
    if not Battle(hero, monster).run():
        break
else:
    print(f"\n🎊 恭喜通關！{hero.name} Lv.{hero.level}")
```

## 這個專案用到了什麼？

| 概念 | 在哪裡 |
|---|---|
| 抽象類別 | `Character` 規定所有角色都要有 `act()` |
| 繼承 + `super()` | `Hero`、`Slime`、`Dragon` 共用 `Character` 的血量和受傷邏輯 |
| 多型 | `Battle` 只呼叫 `act()`，不用管對方是英雄、史萊姆還是龍 |
| property | `hp` 自動限制在 0 ～ 最大值之間，`alive` 是計算屬性 |
| 魔術方法 | `__str__` 顯示血條 |
| dataclass | `Item` 道具 |
| 類別屬性 | `exp_reward` 每種怪物不同 |

class Character:
    def __init__(self, name, hp, attack, defense):
        self.name, self.hp, self.attack, self.defense = name, hp, attack, defense

    def take_damage(self, raw):
        dmg = max(1, raw - self.defense)
        self.hp = max(0, self.hp - dmg)
        return dmg


class Monster(Character):
    exp_reward = 10

    def act(self, target):
        dmg = target.take_damage(self.attack)
        return f"👊 {self.name} 攻擊，造成 {dmg} 點傷害"


class Archer(Monster):
    exp_reward = 20

    def __init__(self):
        super().__init__("弓箭手", 40, 9, 2)

    def act(self, target):
        total = target.take_damage(self.attack) + target.take_damage(self.attack)
        return f"🏹 {self.name} 連射兩箭，造成 {total} 點傷害"
