**魔術方法（magic methods / dunder methods）** 是前後都有兩個底線的特殊方法，例如 `__init__`。定義它們，可以讓你的物件**支援 Python 的內建語法**：`print()`、`+`、`==`、`len()`、`for`、`[]`…

## `__str__` 與 `__repr__`：物件怎麼顯示

```python
class Point:
    def __init__(self, x, y):
        self.x, self.y = x, y

p = Point(3, 4)
print(p)          # 預設的顯示方式很難看
```

```python
class Point:
    def __init__(self, x, y):
        self.x, self.y = x, y

    def __str__(self):            # 給「使用者」看的：print()、str()、f-string
        return f"({self.x}, {self.y})"

    def __repr__(self):           # 給「開發者」看的：除錯、互動模式、放在串列裡
        return f"Point({self.x}, {self.y})"

p = Point(3, 4)
print(p)
print(f"點的位置：{p}")
print([p, Point(0, 0)])           # 串列裡的元素用 __repr__
print(repr(p))
```

> 💡 如果只寫一個，就寫 `__repr__`（`__str__` 沒定義時會自動使用 `__repr__`）。好的 `__repr__` 最好可以直接複製回去建立物件。

## 運算子：`__add__`、`__eq__`、`__lt__`…

```python
class Vector:
    def __init__(self, x, y):
        self.x, self.y = x, y

    def __repr__(self):
        return f"Vector({self.x}, {self.y})"

    def __add__(self, other):         # v1 + v2
        return Vector(self.x + other.x, self.y + other.y)

    def __sub__(self, other):         # v1 - v2
        return Vector(self.x - other.x, self.y - other.y)

    def __mul__(self, k):             # v * 3
        return Vector(self.x * k, self.y * k)

    __rmul__ = __mul__                # 3 * v（數字在左邊）

    def __eq__(self, other):          # v1 == v2
        return (self.x, self.y) == (other.x, other.y)

    def __abs__(self):                # abs(v)：向量長度
        return (self.x ** 2 + self.y ** 2) ** 0.5

a, b = Vector(1, 2), Vector(3, 4)
print(a + b, b - a, a * 3, 3 * a)
print(a == Vector(1, 2), abs(b))
```

| 方法 | 對應的語法 |
|---|---|
| `__add__` `__sub__` `__mul__` `__truediv__` | `+` `-` `*` `/` |
| `__eq__` `__ne__` | `==` `!=` |
| `__lt__` `__le__` `__gt__` `__ge__` | `<` `<=` `>` `>=` |
| `__len__` | `len(x)` |
| `__getitem__` `__setitem__` | `x[key]`、`x[key] = v` |
| `__contains__` | `item in x` |
| `__iter__` | `for item in x` |
| `__call__` | `x()` 把物件當函式呼叫 |
| `__bool__` | `if x:` |

## 比較與排序

定義 `__lt__` 之後，物件就可以用 `sorted()`、`min()`、`max()`：

```python
from functools import total_ordering

@total_ordering                      # 只要寫 __eq__ 和 __lt__，其他比較自動產生
class Version:
    def __init__(self, text):
        self.parts = tuple(int(p) for p in text.split("."))

    def __repr__(self):
        return ".".join(map(str, self.parts))

    def __eq__(self, other):
        return self.parts == other.parts

    def __lt__(self, other):
        return self.parts < other.parts

vs = [Version("1.10.0"), Version("1.2.3"), Version("1.9"), Version("2.0")]
print(sorted(vs))                    # 注意 1.10 > 1.9（用字串比就會錯）
print(max(vs), Version("1.2") <= Version("1.10"))
```

## 容器類：`__len__`、`__getitem__`、`__contains__`

```python
class Playlist:
    def __init__(self, name, songs):
        self.name = name
        self._songs = list(songs)

    def __len__(self):
        return len(self._songs)

    def __getitem__(self, i):         # 支援 [] 和切片，也自動支援 for 迴圈！
        return self._songs[i]

    def __contains__(self, song):
        return song in self._songs

    def __bool__(self):
        return len(self) > 0

p = Playlist("讀書用", ["晴天", "稻香", "七里香", "告白氣球"])
print(len(p), p[0], p[-1], p[1:3])
print("稻香" in p)
for song in p:
    print("♪", song)
print(bool(Playlist("空的", [])))
```

## `__call__`：讓物件可以被呼叫

```python
class Multiplier:
    def __init__(self, k):
        self.k = k

    def __call__(self, x):
        return x * self.k

double = Multiplier(2)
print(double(21))                    # 物件像函式一樣使用
print(list(map(Multiplier(10), [1, 2, 3])))
```

> 💡 PyTorch 的神經網路模型就是用 `__call__`：你寫 `model(x)`，實際上執行的是模型的 `forward` 方法。

class Money:
    def __init__(self, amount, currency="TWD"):
        self.amount = amount
        self.currency = currency

    def __repr__(self):
        return f"Money({self.amount}, {self.currency!r})"

    def __str__(self):
        return f"{self.currency} {self.amount:,}"

    def __add__(self, other):
        if self.currency != other.currency:
            raise ValueError("貨幣不同，不能相加")
        return Money(self.amount + other.amount, self.currency)

    def __eq__(self, other):
        return (self.amount, self.currency) == (other.amount, other.currency)

    def __lt__(self, other):
        return self.amount < other.amount


a = Money(1000)
b = Money(234)
print(a + b)
print(repr(a), a == Money(1000), b < a)
