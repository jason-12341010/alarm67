## 樹（Tree）

樹是**階層式**的資料結構：資料夾、公司組織圖、HTML 的結構、決策樹（第 8 階段的機器學習會用到）。

```text
        8          ← 根節點（root）
       / \
      3   10
     / \    \
    1   6    14    ← 沒有子節點的叫葉節點（leaf）
       / \   /
      4   7 13
```

## 二元搜尋樹（BST）

每個節點：**左邊的都比它小，右邊的都比它大**。查詢、插入平均 O(log n)：

```python
class Node:
    def __init__(self, value):
        self.value = value
        self.left = None
        self.right = None

class BST:
    def __init__(self):
        self.root = None

    def insert(self, value):
        if self.root is None:
            self.root = Node(value)
            return
        node = self.root
        while True:
            if value < node.value:
                if node.left is None:
                    node.left = Node(value); return
                node = node.left
            else:
                if node.right is None:
                    node.right = Node(value); return
                node = node.right

    def contains(self, value):
        node = self.root
        while node:
            if value == node.value:
                return True
            node = node.left if value < node.value else node.right
        return False

tree = BST()
for v in [8, 3, 10, 1, 6, 14, 4, 7, 13]:
    tree.insert(v)
print(tree.contains(6), tree.contains(5))
```

## 走訪樹：遞迴最適合

```python
class Node:
    def __init__(self, value, left=None, right=None):
        self.value, self.left, self.right = value, left, right

#        8
#       / \
#      3   10
#     / \    \
#    1   6    14
root = Node(8, Node(3, Node(1), Node(6)), Node(10, None, Node(14)))

def inorder(n):          # 中序：左 → 自己 → 右（BST 會得到排序好的結果！）
    return inorder(n.left) + [n.value] + inorder(n.right) if n else []

def preorder(n):         # 前序：自己 → 左 → 右
    return [n.value] + preorder(n.left) + preorder(n.right) if n else []

def height(n):
    return 1 + max(height(n.left), height(n.right)) if n else 0

def show(n, prefix="", is_left=True):
    if n:
        show(n.right, prefix + ("│   " if is_left else "    "), False)
        print(prefix + ("└── " if is_left else "┌── ") + str(n.value))
        show(n.left, prefix + ("    " if is_left else "│   "), True)

print("中序：", inorder(root))
print("前序：", preorder(root))
print("高度：", height(root))
show(root)
```

## 圖（Graph）

圖由**節點**和**邊**組成，可以表示任何「關係」：社群網路的好友、地圖上的道路、網頁之間的連結、捷運路線…

最常用**鄰接串列（adjacency list）** 表示，在 Python 裡就是一個字典：

```python
graph = {
    "台北": ["板橋", "松山", "中山"],
    "板橋": ["台北", "新莊"],
    "松山": ["台北", "南港"],
    "中山": ["台北", "淡水"],
    "新莊": ["板橋"],
    "南港": ["松山"],
    "淡水": ["中山"],
}
for station, neighbors in graph.items():
    print(f"{station} ↔ {'、'.join(neighbors)}")
```

## 廣度優先搜尋（BFS）：找最短路徑

用**佇列**，從起點開始**一層一層**往外擴散。在沒有權重的圖上，BFS 第一次找到終點時，就是**最少步數**的路徑：

```python
from collections import deque

graph = {
    "台北": ["板橋", "松山", "中山"], "板橋": ["台北", "新莊"], "松山": ["台北", "南港"],
    "中山": ["台北", "淡水"], "新莊": ["板橋"], "南港": ["松山"], "淡水": ["中山"],
}

def shortest_path(graph, start, goal):
    queue = deque([[start]])
    visited = {start}
    while queue:
        path = queue.popleft()
        node = path[-1]
        if node == goal:
            return path
        for nxt in graph[node]:
            if nxt not in visited:
                visited.add(nxt)
                queue.append(path + [nxt])
    return None

print(" → ".join(shortest_path(graph, "新莊", "淡水")))
print(" → ".join(shortest_path(graph, "南港", "新莊")))
```

## 深度優先搜尋（DFS）：一路走到底

用**遞迴**（或堆疊），先一條路走到底，走不通再回頭。適合「找出所有可能」、「判斷是否連通」：

```python
grid = [
    "11000",
    "11010",
    "00011",
    "10000",
]

def count_islands(grid):
    rows, cols = len(grid), len(grid[0])
    seen = set()

    def dfs(r, c):
        if not (0 <= r < rows and 0 <= c < cols) or grid[r][c] == "0" or (r, c) in seen:
            return
        seen.add((r, c))
        for dr, dc in [(1, 0), (-1, 0), (0, 1), (0, -1)]:   # 上下左右
            dfs(r + dr, c + dc)

    islands = 0
    for r in range(rows):
        for c in range(cols):
            if grid[r][c] == "1" and (r, c) not in seen:
                dfs(r, c)                # 把整座島都標記起來
                islands += 1
    return islands

print("島嶼數量：", count_islands(grid))
```

## 有權重的圖：Dijkstra 演算法

道路有不同的距離時，用 **Dijkstra** 找最短距離（搭配優先佇列）：

```python
import heapq

roads = {
    "A": {"B": 4, "C": 2},
    "B": {"C": 5, "D": 10},
    "C": {"E": 3},
    "D": {"F": 11},
    "E": {"D": 4},
    "F": {},
}

def dijkstra(graph, start):
    dist = {start: 0}
    pq = [(0, start)]
    while pq:
        d, node = heapq.heappop(pq)
        if d > dist.get(node, float("inf")):
            continue
        for nxt, w in graph[node].items():
            nd = d + w
            if nd < dist.get(nxt, float("inf")):
                dist[nxt] = nd
                heapq.heappush(pq, (nd, nxt))
    return dist

print(dijkstra(roads, "A"))
```

Google 地圖的導航，背後就是這類演算法的進化版。

> 💡 實務上處理圖可以用 **networkx** 套件：`nx.shortest_path(G, "A", "F", weight="weight")`。

from collections import deque

friends = {
    "小明": ["小美", "阿華"],
    "小美": ["小明", "小強"],
    "阿華": ["小明"],
    "小強": ["小美", "小芳"],
    "小芳": ["小強"],
    "獨行俠": [],
}


def degrees(friends, a, b):
    queue = deque([(a, 0)])
    visited = {a}
    while queue:
        person, d = queue.popleft()
        if person == b:
            return d
        for f in friends.get(person, []):
            if f not in visited:
                visited.add(f)
                queue.append((f, d + 1))
    return -1


print(degrees(friends, "阿華", "小芳"))
print(degrees(friends, "小明", "獨行俠"))
