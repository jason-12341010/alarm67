這是資料科學界最有名的入門專案（Kaggle 的 Titanic 比賽）：**根據乘客資料，預測誰能在鐵達尼號沉船中生還**。我們要把整個第 8 階段學的東西全部用上。

`titanic.csv` 是真實的歷史資料，891 位乘客：

| 欄位 | 意思 |
|---|---|
| Survived | 是否生還（0 = 否、1 = 是）← **要預測的目標** |
| Pclass | 艙等（1 = 頭等、2 = 二等、3 = 三等） |
| Name、Sex、Age | 姓名、性別、年齡 |
| SibSp / Parch | 同行的兄弟姊妹配偶 / 父母子女人數 |
| Ticket、Fare、Cabin | 船票號碼、票價、艙房號碼 |
| Embarked | 登船港口 |

def make_features(df):

## Step 4：建立 Pipeline 並比較模型

```python
import pandas as pd
from sklearn.pipeline import Pipeline
from sklearn.compose import ColumnTransformer
from sklearn.impute import SimpleImputer
from sklearn.preprocessing import OneHotEncoder, StandardScaler
from sklearn.model_selection import cross_val_score, StratifiedKFold
from sklearn.linear_model import LogisticRegression
from sklearn.ensemble import RandomForestClassifier, HistGradientBoostingClassifier

def make_features(df):
    df = df.copy()
    df["Title"] = df["Name"].str.extract(r",\s*([^.]+)\.")[0].replace({"Mlle": "Miss", "Ms": "Miss", "Mme": "Mrs"})
    df.loc[~df["Title"].isin(["Mr", "Mrs", "Miss", "Master"]), "Title"] = "Rare"
    df["FamilySize"] = df["SibSp"] + df["Parch"] + 1
    df["IsAlone"] = (df["FamilySize"] == 1).astype(int)
    df["HasCabin"] = df["Cabin"].notna().astype(int)
    return df

df = make_features(pd.read_csv("titanic.csv"))
numeric = ["Age", "Fare", "FamilySize", "SibSp", "Parch"]
categorical = ["Pclass", "Sex", "Embarked", "Title", "IsAlone", "HasCabin"]
X, y = df[numeric + categorical], df["Survived"]

prep = ColumnTransformer([
    ("num", Pipeline([("imp", SimpleImputer(strategy="median")), ("sc", StandardScaler())]), numeric),
    ("cat", Pipeline([("imp", SimpleImputer(strategy="most_frequent")), ("oh", OneHotEncoder(handle_unknown="ignore"))]), categorical),
])
cv = StratifiedKFold(5, shuffle=True, random_state=42)
for name, clf in {
    "邏輯迴歸": LogisticRegression(max_iter=1000),
    "隨機森林": RandomForestClassifier(n_estimators=200, max_depth=6, random_state=0),
    "梯度提升": HistGradientBoostingClassifier(max_depth=3, learning_rate=0.05, random_state=0),
}.items():
    s = cross_val_score(Pipeline([("prep", prep), ("clf", clf)]), X, y, cv=cv)
    print(f"{name:<6} {s.mean():.3f} ± {s.std():.3f}")
```

## Step 5：檢視錯誤與特徵重要性

```python
import pandas as pd
import matplotlib.pyplot as plt
from sklearn.pipeline import Pipeline
from sklearn.compose import ColumnTransformer
from sklearn.impute import SimpleImputer
from sklearn.preprocessing import OneHotEncoder
from sklearn.ensemble import RandomForestClassifier
from sklearn.model_selection import train_test_split
from sklearn.metrics import classification_report

df = pd.read_csv("titanic.csv")
df["Title"] = df["Name"].str.extract(r",\s*([^.]+)\.")[0]
df.loc[~df["Title"].isin(["Mr", "Mrs", "Miss", "Master"]), "Title"] = "Rare"
df["FamilySize"] = df["SibSp"] + df["Parch"] + 1
numeric, categorical = ["Age", "Fare", "FamilySize"], ["Pclass", "Sex", "Title"]
prep = ColumnTransformer([
    ("num", SimpleImputer(strategy="median"), numeric),
    ("cat", OneHotEncoder(handle_unknown="ignore"), categorical),
])
model = Pipeline([("prep", prep), ("rf", RandomForestClassifier(n_estimators=200, max_depth=6, random_state=0))])
X_train, X_test, y_train, y_test = train_test_split(df[numeric + categorical], df["Survived"], test_size=0.2, random_state=0, stratify=df["Survived"])
model.fit(X_train, y_train)
print(classification_report(y_test, model.predict(X_test), target_names=["死亡", "生還"]))

names = model["prep"].get_feature_names_out()
imp = pd.Series(model["rf"].feature_importances_, index=names).sort_values()
imp.tail(10).plot(kind="barh", figsize=(7, 4), title="特徵重要性（前 10 名）")
plt.show()
```

## Step 6：預測你自己！

```python
import pandas as pd
from sklearn.pipeline import Pipeline
from sklearn.compose import ColumnTransformer
from sklearn.impute import SimpleImputer
from sklearn.preprocessing import OneHotEncoder
from sklearn.ensemble import RandomForestClassifier

df = pd.read_csv("titanic.csv")
features = ["Pclass", "Sex", "Age", "SibSp", "Parch", "Fare"]
prep = ColumnTransformer([
    ("num", SimpleImputer(strategy="median"), ["Age", "SibSp", "Parch", "Fare"]),
    ("cat", OneHotEncoder(handle_unknown="ignore"), ["Pclass", "Sex"]),
])
model = Pipeline([("prep", prep), ("rf", RandomForestClassifier(n_estimators=200, max_depth=6, random_state=0))])
model.fit(df[features], df["Survived"])

me = pd.DataFrame([
    {"Pclass": 3, "Sex": "male", "Age": 22, "SibSp": 0, "Parch": 0, "Fare": 8},
    {"Pclass": 1, "Sex": "female", "Age": 30, "SibSp": 1, "Parch": 0, "Fare": 100},
    {"Pclass": 2, "Sex": "male", "Age": 6, "SibSp": 1, "Parch": 2, "Fare": 30},
])
for (_, p), prob in zip(me.iterrows(), model.predict_proba(me)[:, 1]):
    print(f"{p.Pclass} 等艙、{'女' if p.Sex == 'female' else '男'}、{p.Age} 歲 → 生還機率 {prob:.0%}")
```
