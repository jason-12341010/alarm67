```text

  ↓  pypdf 讀取每一頁的文字
頁面文字
  ↓  依句子切塊（記住頁碼）

```

## Part 1：在網頁上做出核心邏輯

先用**模擬的文件**把「切塊 → 建立索引 → 檢索 → 組合提示詞 → 整理引用」全部做出來。這部分可以直接執行：

```python
import re
from dataclasses import dataclass, field
import numpy as np
from sklearn.feature_extraction.text import TfidfVectorizer

# 模擬：一份 PDF 的每一頁文字
PAGES = [
    "Python 學習指南\n第一章：認識 Python。Python 由 Guido van Rossum 在 1991 年發表。它的設計哲學強調程式碼的可讀性。",
    "第二章：資料型別。Python 的基本型別有 int、float、str、bool。串列 list 可以修改，元組 tuple 不能修改。",
    "第三章：函式。用 def 定義函式，用 return 回傳結果。函式可以有預設參數，也可以用 *args 接收任意數量的參數。",
    "第四章：物件導向。class 用來定義類別。__init__ 是建構子，在建立物件時自動執行。繼承讓子類別可以重複使用父類別的程式。",
    "第五章：資料分析。pandas 提供 DataFrame 處理表格資料。NumPy 提供高效能的陣列運算，是科學計算的基礎。",
]


@dataclass
class Chunk:
    text: str
    page: int


@dataclass
class PdfQA:
    chunk_size: int = 60
    chunks: list = field(default_factory=list)

    def add_pages(self, pages):
        for page_no, text in enumerate(pages, start=1):
            sentences = [s.strip() for s in re.split(r"(?<=[。！？\n])", text) if s.strip()]
            current = ""
            for s in sentences:
                if len(current) + len(s) > self.chunk_size and current:
                    self.chunks.append(Chunk(current, page_no))
                    current = ""
                current += s
            if current:
                self.chunks.append(Chunk(current, page_no))
        self._build_index()

    def _build_index(self):
        self.vectorizer = TfidfVectorizer(analyzer="char", ngram_range=(1, 2))
        m = self.vectorizer.fit_transform([c.text for c in self.chunks]).toarray()
        self.matrix = m / (np.linalg.norm(m, axis=1, keepdims=True) + 1e-9)

    def search(self, question, k=3):
        q = self.vectorizer.transform([question]).toarray()[0]
        q = q / (np.linalg.norm(q) + 1e-9)
        sims = self.matrix @ q
        return [(self.chunks[i], float(sims[i])) for i in np.argsort(-sims)[:k]]

    def build_prompt(self, question, k=3):
        hits = self.search(question, k)
        docs = "\n".join(f'<source page="{c.page}">{c.text}</source>' for c, _ in hits)
        return (
            "請只根據下面的資料回答問題。資料中沒有答案時，請回答「文件中沒有提到」。\n"
            "每個重點後面用 [第 N 頁] 標註出處。\n\n"
            f"<sources>\n{docs}\n</sources>\n\n問題：{question}"
        )


qa = PdfQA()
qa.add_pages(PAGES)
print(f"共切成 {len(qa.chunks)} 塊")
for c, score in qa.search("tuple 可以修改嗎？"):
    print(f"  {score:.3f}  [第 {c.page} 頁] {c.text}")
print("\n" + qa.build_prompt("tuple 可以修改嗎？"))
```

### 程式：`pdf_qa.py`

```py
import re
from dataclasses import dataclass

import anthropic
import gradio as gr
import numpy as np
from pypdf import PdfReader
from sentence_transformers import SentenceTransformer

client = anthropic.Anthropic()
embedder = SentenceTransformer("paraphrase-multilingual-MiniLM-L12-v2")

SYSTEM = """你是文件問答助理。
- 只根據 <sources> 裡的資料回答，不要使用外部知識。
- 資料中沒有答案時，回答「文件中沒有提到」。
- 每個重點後面用 [第 N 頁] 標註出處。
- 使用繁體中文，簡潔清楚。"""


@dataclass
class Chunk:
    text: str
    page: int


class PdfQA:
    def __init__(self, chunk_size=300):
        self.chunk_size = chunk_size
        self.chunks: list[Chunk] = []
        self.vectors = None

    def load_pdf(self, path):
        self.chunks = []
        for page_no, page in enumerate(PdfReader(path).pages, start=1):
            text = page.extract_text() or ""
            sentences = [s.strip() for s in re.split(r"(?<=[。！？.!?\n])", text) if s.strip()]
            current = ""
            for s in sentences:
                if len(current) + len(s) > self.chunk_size and current:
                    self.chunks.append(Chunk(current, page_no))
                    current = ""
                current += s
            if current:
                self.chunks.append(Chunk(current, page_no))
        self.vectors = embedder.encode([c.text for c in self.chunks], normalize_embeddings=True)
        return len(self.chunks)

    def search(self, question, k=5):
        q = embedder.encode([question], normalize_embeddings=True)[0]
        return [self.chunks[i] for i in np.argsort(-(self.vectors @ q))[:k]]

    def ask(self, question, history):
        sources = "\n".join(f'<source page="{c.page}">{c.text}</source>' for c in self.search(question))
        messages = [{"role": m["role"], "content": m["content"]} for m in history]
        messages.append({"role": "user", "content": f"<sources>\n{sources}\n</sources>\n\n問題：{question}"})
        reply = ""
        with client.messages.stream(model="claude-opus-5-5", max_tokens=2048,
                                    system=SYSTEM, messages=messages) as stream:
            for text in stream.text_stream:
                reply += text
                yield reply


qa = PdfQA()

with gr.Blocks(title="PDF 問答機器人") as demo:
    gr.Markdown("# 📄 PDF 問答機器人\n上傳 PDF，然後開始發問！")
    pdf = gr.File(label="上傳 PDF", file_types=[".pdf"])
    status = gr.Markdown()
    pdf.upload(lambda f: f"✅ 已讀取，共 {qa.load_pdf(f)} 個段落", pdf, status)
    gr.ChatInterface(qa.ask, type="messages")

demo.launch()
```

import re


def extract_citations(answer):
    pages = set()
    for group in re.findall(r"\[第\s*([\d、,\s]+?)\s*頁\]", answer):
        pages.update(int(n) for n in re.findall(r"\d+", group))
    return sorted(pages)
