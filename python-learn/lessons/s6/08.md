**SQLAlchemy** 是 Python 最強大的資料庫工具，也是 Flask 和 FastAPI 最常搭配的 **ORM**。它支援 SQLite、PostgreSQL、MySQL 等各種資料庫，換資料庫幾乎不用改程式。

好消息：SQLAlchemy **可以在這個網站直接執行**（搭配 SQLite）！

## 為什麼用 ORM？

```text
直接寫 SQL：   cur.execute("SELECT * FROM users WHERE age > ?", (18,))
               → 拿到一堆 tuple，要自己記得每個位置是什麼

用 ORM：        session.scalars(select(User).where(User.age > 18))
               → 拿到 User 物件，可以用 user.name、user.orders…
```

- ✅ 用 Python 類別描述資料表，有型別提示，編輯器會自動完成
- ✅ 自動防止 SQL 注入
- ✅ 資料表之間的關聯變得很直覺
- ⚠️ 複雜的查詢還是要懂 SQL，ORM 不能取代 SQL 知識

## 定義資料表（SQLAlchemy 2.0 的寫法）

```python
from sqlalchemy import create_engine, String, ForeignKey, select
from sqlalchemy.orm import DeclarativeBase, Mapped, mapped_column, relationship, Session


class Base(DeclarativeBase):
    pass


class User(Base):
    __tablename__ = "users"
    id: Mapped[int] = mapped_column(primary_key=True)
    name: Mapped[str] = mapped_column(String(50))
    email: Mapped[str] = mapped_column(String(100), unique=True)
    orders: Mapped[list["Order"]] = relationship(back_populates="user")

    def __repr__(self):
        return f"User({self.name!r})"


class Order(Base):
    __tablename__ = "orders"
    id: Mapped[int] = mapped_column(primary_key=True)
    product: Mapped[str]
    amount: Mapped[int]
    user_id: Mapped[int] = mapped_column(ForeignKey("users.id"))
    user: Mapped[User] = relationship(back_populates="orders")


engine = create_engine("sqlite:///:memory:", echo=False)   # 實際上：sqlite:///app.db 或 postgresql://...
Base.metadata.create_all(engine)                          # 建立所有資料表

with Session(engine) as session:
    ming = User(name="小明", email="ming@example.com")
    mei = User(name="小美", email="mei@example.com")
    ming.orders = [Order(product="耳機", amount=1290), Order(product="滑鼠", amount=590)]
    mei.orders.append(Order(product="鍵盤", amount=2490))
    session.add_all([ming, mei])
    session.commit()

    for user in session.scalars(select(User)):
        print(user.name, "→", [(o.product, o.amount) for o in user.orders])
```

> 💡 `echo=True` 會印出 SQLAlchemy 實際執行的 SQL，學習時很有幫助。改成 True 再執行一次看看！

## 查詢

```python
from sqlalchemy import create_engine, String, ForeignKey, select, func
from sqlalchemy.orm import DeclarativeBase, Mapped, mapped_column, relationship, Session

class Base(DeclarativeBase): pass

class Product(Base):
    __tablename__ = "products"
    id: Mapped[int] = mapped_column(primary_key=True)
    name: Mapped[str]
    category: Mapped[str]
    price: Mapped[int]
    stock: Mapped[int] = mapped_column(default=0)

engine = create_engine("sqlite:///:memory:")
Base.metadata.create_all(engine)

with Session(engine) as s:
    s.add_all([
        Product(name="藍牙耳機", category="3C", price=1290, stock=15),
        Product(name="無線滑鼠", category="3C", price=590, stock=0),
        Product(name="機械鍵盤", category="3C", price=2490, stock=8),
        Product(name="咖啡豆", category="食品", price=520, stock=30),
        Product(name="保溫瓶", category="居家", price=790, stock=12),
    ])
    s.commit()

    print("1000 元以下、有庫存：")
    stmt = select(Product).where(Product.price <= 1000, Product.stock > 0).order_by(Product.price)
    for p in s.scalars(stmt):
        print(f"  {p.name} {p.price}")

    print("名稱有「機」：", [p.name for p in s.scalars(select(Product).where(Product.name.contains("機")))])

    p = s.get(Product, 3)                       # 用主鍵查
    print("編號 3：", p.name)

    print("各類別的商品數和平均價格：")
    stmt = select(Product.category, func.count(), func.avg(Product.price)).group_by(Product.category)
    for cat, n, avg in s.execute(stmt):
        print(f"  {cat}：{n} 項，平均 {avg:.0f} 元")
```

## 修改與刪除

```python
from sqlalchemy import create_engine, select, update, delete
from sqlalchemy.orm import DeclarativeBase, Mapped, mapped_column, Session

class Base(DeclarativeBase): pass

class Product(Base):
    __tablename__ = "products"
    id: Mapped[int] = mapped_column(primary_key=True)
    name: Mapped[str]
    price: Mapped[int]

engine = create_engine("sqlite:///:memory:")
Base.metadata.create_all(engine)

with Session(engine) as s:
    s.add_all([Product(name="A", price=100), Product(name="B", price=200), Product(name="C", price=300)])
    s.commit()

    p = s.scalars(select(Product).where(Product.name == "A")).one()
    p.price = 150                               # 直接改屬性
    s.execute(update(Product).where(Product.price >= 200).values(price=Product.price * 0.9))   # 批次修改
    s.delete(s.get(Product, 2))                 # 刪除一筆
    s.commit()
    print([(p.name, p.price) for p in s.scalars(select(Product))])
```

## Session：工作單位

`Session` 會記住你所有的修改，`commit()` 時**一次寫入**；中間出錯可以 `rollback()` 全部取消（第 4 階段的交易概念）：

```python
from sqlalchemy import create_engine, select
from sqlalchemy.orm import DeclarativeBase, Mapped, mapped_column, Session

class Base(DeclarativeBase): pass

class Account(Base):
    __tablename__ = "accounts"
    id: Mapped[int] = mapped_column(primary_key=True)
    owner: Mapped[str]
    balance: Mapped[int]

engine = create_engine("sqlite:///:memory:")
Base.metadata.create_all(engine)
with Session(engine) as s:
    s.add_all([Account(owner="小明", balance=1000), Account(owner="小美", balance=500)])
    s.commit()

def transfer(session, from_id, to_id, amount):
    a, b = session.get(Account, from_id), session.get(Account, to_id)
    a.balance -= amount
    b.balance += amount
    if a.balance < 0:
        raise ValueError("餘額不足")

with Session(engine) as s:
    for amount in [300, 5000]:
        try:
            transfer(s, 1, 2, amount)
            s.commit()
            print(f"✅ 轉帳 {amount} 成功")
        except ValueError as e:
            s.rollback()                         # 全部還原
            print(f"❌ 轉帳 {amount} 失敗：{e}")
    print([(a.owner, a.balance) for a in s.scalars(select(Account))])
```

## 搭配 FastAPI

```py
from fastapi import FastAPI, Depends, HTTPException
from sqlalchemy import create_engine, select
from sqlalchemy.orm import Session
from models import Base, Product           # 上面定義的 Model

engine = create_engine("sqlite:///shop.db")
Base.metadata.create_all(engine)
app = FastAPI()

def get_session():                          # 依賴注入：每個請求一個 Session
    with Session(engine) as session:
        yield session

@app.get("/products/{pid}")
def get_product(pid: int, session: Session = Depends(get_session)):
    product = session.get(Product, pid)
    if product is None:
        raise HTTPException(404, "找不到商品")
    return {"id": product.id, "name": product.name, "price": product.price}
```

from sqlalchemy import create_engine, select
from sqlalchemy.orm import DeclarativeBase, Mapped, mapped_column, Session


class Base(DeclarativeBase):
    pass


class Book(Base):
    __tablename__ = "books"
    id: Mapped[int] = mapped_column(primary_key=True)
    title: Mapped[str]
    author: Mapped[str]
    year: Mapped[int]
    available: Mapped[bool] = mapped_column(default=True)


engine = create_engine("sqlite:///:memory:")
Base.metadata.create_all(engine)


def add_books(session, books):
    session.add_all([Book(**b) for b in books])
    session.commit()


def borrow(session, book_id):
    book = session.get(Book, book_id)
    if book is None or not book.available:
        return False
    book.available = False
    session.commit()
    return True


def available_by_author(session, author):
    stmt = select(Book.title).where(Book.author == author, Book.available).order_by(Book.year)
    return list(session.scalars(stmt))


with Session(engine) as s:
    add_books(s, [
        {"title": "挪威的森林", "author": "村上春樹", "year": 1987},

    ])

    print(available_by_author(s, "村上春樹"))
