網站要能**接收使用者的資料**（表單），並且**存起來**（資料庫）。這一課做一個可以新增、刪除的**留言板**。

## 資料庫：SQLite

**SQLite** 是一個**檔案型**的資料庫，Python 內建 `sqlite3` 模組，不用安裝任何東西。很適合小型網站和學習。資料庫用 **SQL** 語言操作，在網頁上就可以練習：

```python
import sqlite3

conn = sqlite3.connect(":memory:")      # 記憶體中的資料庫（練習用）；實際上用檔名，例如 "app.db"
cur = conn.cursor()

cur.execute("""
    CREATE TABLE messages (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        author TEXT NOT NULL,
        content TEXT NOT NULL,
        created_at TEXT DEFAULT CURRENT_TIMESTAMP
    )
""")

cur.execute("INSERT INTO messages (author, content) VALUES (?, ?)", ("小明", "大家好！"))
cur.executemany("INSERT INTO messages (author, content) VALUES (?, ?)", [
    ("小美", "今天天氣真好"),
    ("阿華", "有人要一起學 Python 嗎？"),
])
conn.commit()                            # 確認寫入

for row in cur.execute("SELECT id, author, content FROM messages ORDER BY id DESC"):
    print(row)
```

### SQL 基本指令（CRUD）

| 操作 | SQL |
|---|---|
| **C**reate 新增 | `INSERT INTO 表格 (欄位) VALUES (?, ?)` |
| **R**ead 查詢 | `SELECT 欄位 FROM 表格 WHERE 條件 ORDER BY 欄位` |
| **U**pdate 修改 | `UPDATE 表格 SET 欄位 = ? WHERE id = ?` |
| **D**elete 刪除 | `DELETE FROM 表格 WHERE id = ?` |

```python
import sqlite3

conn = sqlite3.connect(":memory:")
conn.row_factory = sqlite3.Row           # 查詢結果可以用欄位名稱取值
conn.execute("CREATE TABLE products (id INTEGER PRIMARY KEY, name TEXT, price INTEGER, stock INTEGER)")
conn.executemany("INSERT INTO products (name, price, stock) VALUES (?, ?, ?)", [
    ("藍牙耳機", 1290, 15), ("無線滑鼠", 590, 0), ("機械鍵盤", 2490, 8), ("USB 線", 290, 50),
])

print("有庫存且 1000 元以下：")
for r in conn.execute("SELECT * FROM products WHERE stock > 0 AND price <= ? ORDER BY price", (1000,)):
    print(f"  {r['name']}：{r['price']} 元（庫存 {r['stock']}）")

conn.execute("UPDATE products SET stock = stock - 1 WHERE name = ?", ("機械鍵盤",))
conn.execute("DELETE FROM products WHERE stock = 0")
print("總庫存價值：", conn.execute("SELECT SUM(price * stock) FROM products").fetchone()[0])
print("剩下幾種商品：", conn.execute("SELECT COUNT(*) FROM products").fetchone()[0])
```

## ⚠️ SQL 注入：永遠用 `?` 傳參數

**絕對不要**用字串組合 SQL！使用者可以輸入惡意的內容，竊取或刪除你的資料：

```python
import sqlite3

conn = sqlite3.connect(":memory:")
conn.execute("CREATE TABLE users (name TEXT, password TEXT)")
conn.executemany("INSERT INTO users VALUES (?, ?)", [("admin", "secret123"), ("amy", "pw456")])

evil = "x' OR '1'='1"                     # 駭客輸入的「名字」

# ❌ 危險：用 f-string 組合 SQL
sql = f"SELECT * FROM users WHERE name = '{evil}'"
print("組出來的 SQL：", sql)
print("❌ 撈出了：", conn.execute(sql).fetchall())      # 全部的帳號密碼都外洩了！

# ✅ 安全：用 ? 參數，資料庫會把它當成純文字
print("✅ 撈出了：", conn.execute("SELECT * FROM users WHERE name = ?", (evil,)).fetchall())
```

## Flask 表單：留言板

> 📄 以下在自己的電腦執行。專案結構：

```text
guestbook/
├── app.py
└── templates/
    └── index.html
```

`app.py`：

```py
import sqlite3
from flask import Flask, render_template, request, redirect, url_for, flash, g

app = Flask(__name__)
app.secret_key = "change-me"          # flash 訊息需要；正式環境要用隨機的長字串
DB = "guestbook.db"


def get_db():
    if "db" not in g:                  # g：每個請求共用的物件
        g.db = sqlite3.connect(DB)
        g.db.row_factory = sqlite3.Row
    return g.db


@app.teardown_appcontext
def close_db(exc):
    db = g.pop("db", None)
    if db is not None:
        db.close()


def init_db():
    with sqlite3.connect(DB) as conn:
        conn.execute("""CREATE TABLE IF NOT EXISTS messages (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            author TEXT NOT NULL,
            content TEXT NOT NULL,
            created_at TEXT DEFAULT CURRENT_TIMESTAMP)""")


@app.route("/", methods=["GET", "POST"])
def index():
    db = get_db()
    if request.method == "POST":                       # 送出表單
        author = request.form.get("author", "").strip()
        content = request.form.get("content", "").strip()
        if not author or not content:
            flash("名字和內容都要填喔！")
        else:
            db.execute("INSERT INTO messages (author, content) VALUES (?, ?)", (author, content))
            db.commit()
            flash("留言成功 🎉")
        return redirect(url_for("index"))              # 送出後重新導向，避免重新整理時重複送出
    messages = db.execute("SELECT * FROM messages ORDER BY id DESC").fetchall()
    return render_template("index.html", messages=messages)


@app.route("/delete/<int:mid>", methods=["POST"])
def delete(mid):
    db = get_db()
    db.execute("DELETE FROM messages WHERE id = ?", (mid,))
    db.commit()
    return redirect(url_for("index"))


if __name__ == "__main__":
    init_db()
    app.run(debug=True)
```

import sqlite3

conn = sqlite3.connect(":memory:")
conn.execute("CREATE TABLE messages (id INTEGER PRIMARY KEY AUTOINCREMENT, author TEXT, content TEXT)")


def add_message(conn, author, content):
    author, content = author.strip(), content.strip()
    if not author or not content:
        raise ValueError("名字和內容都不能是空的")
    cur = conn.execute("INSERT INTO messages (author, content) VALUES (?, ?)", (author, content))
    conn.commit()
    return cur.lastrowid


def list_messages(conn, keyword=None):
    if keyword:
        rows = conn.execute("SELECT id, author, content FROM messages WHERE content LIKE ? ORDER BY id DESC", (f"%{keyword}%",))
    else:
        rows = conn.execute("SELECT id, author, content FROM messages ORDER BY id DESC")
    return [tuple(r) for r in rows]


def delete_message(conn, mid):
    cur = conn.execute("DELETE FROM messages WHERE id = ?", (mid,))
    conn.commit()
    return cur.rowcount == 1


add_message(conn, "小明", "學 Python 好好玩")
add_message(conn, "小美", "今天下雨")
print(list_messages(conn))
print(list_messages(conn, "Python"))
