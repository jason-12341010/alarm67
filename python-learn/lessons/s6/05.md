**Pydantic** 用**型別提示**定義資料的格式，自動做**驗證**和**轉換**。它是 FastAPI 的核心，也被大量用在設定檔、AI 的結構化輸出等地方。

好消息：Pydantic 可以在這個網站**直接執行**！

## 基本用法

```python
from pydantic import BaseModel

class User(BaseModel):
    name: str
    age: int
    email: str
    is_active: bool = True

u = User(name="小明", age="18", email="ming@example.com")   # "18" 會自動轉成 18
print(u)
print(u.age + 1, type(u.age))
print(u.model_dump())            # 轉成字典
print(u.model_dump_json())       # 轉成 JSON 字串
```

## 驗證失敗時

```python
from pydantic import BaseModel, ValidationError

class User(BaseModel):
    name: str
    age: int
    email: str

try:
    User(name="小美", age="十八")
except ValidationError as e:
    print(f"有 {e.error_count()} 個錯誤：")
    for err in e.errors():
        print(f"  欄位 {err['loc']}：{err['msg']}")
```

一次列出**所有**錯誤，而且清楚說明是哪個欄位。FastAPI 收到不合格的資料時，就是把這些錯誤包成 **422 回應**傳回去。

## Field：更多限制

```python
from pydantic import BaseModel, Field, ValidationError

class Product(BaseModel):
    name: str = Field(min_length=1, max_length=50)
    price: int = Field(gt=0, description="價格（元）")          # gt：大於
    discount: float = Field(default=0, ge=0, le=0.9)           # ge / le：大於等於 / 小於等於
    tags: list[str] = Field(default_factory=list, max_length=5)

print(Product(name="耳機", price=1290, tags=["3C", "藍牙"]))
for bad in [{"name": "", "price": 100}, {"name": "x", "price": -5}, {"name": "x", "price": 10, "discount": 0.95}]:
    try:
        Product(**bad)
    except ValidationError as e:
        print("❌", e.errors()[0]["loc"], e.errors()[0]["msg"])
```

## 常用的特殊型別

```python
from datetime import date
from typing import Literal
from pydantic import BaseModel, EmailStr, HttpUrl, ValidationError

class Signup(BaseModel):
    username: str
    birthday: date                          # "2005-08-17" 自動轉成 date
    plan: Literal["free", "pro"] = "free"   # 只能是這幾個值
    website: HttpUrl | None = None

s = Signup(username="amy", birthday="2005-08-17", website="https://amy.dev")
print(s.birthday.year, s.plan, s.website.host)
try:
    Signup(username="bob", birthday="2005-02-30", plan="vip")
except ValidationError as e:
    for err in e.errors():
        print("❌", err["loc"], err["msg"])
```

> 💡 `EmailStr` 需要另外安裝 `email-validator` 套件（`pip install "pydantic[email]"`）。

## 自訂驗證

```python
from pydantic import BaseModel, field_validator, model_validator, ValidationError

class Register(BaseModel):
    username: str
    password: str
    password2: str

    @field_validator("username")
    @classmethod
    def no_space(cls, v: str) -> str:
        if " " in v:
            raise ValueError("使用者名稱不能有空白")
        return v.lower()                       # 也可以順便轉換

    @field_validator("password")
    @classmethod
    def strong(cls, v: str) -> str:
        if len(v) < 8 or not any(c.isdigit() for c in v):
            raise ValueError("密碼至少 8 個字，而且要有數字")
        return v

    @model_validator(mode="after")
    def passwords_match(self):
        if self.password != self.password2:
            raise ValueError("兩次密碼不一樣")
        return self

print(Register(username="Amy", password="abc12345", password2="abc12345"))
for data in [
    {"username": "a b", "password": "abc12345", "password2": "abc12345"},
    {"username": "amy", "password": "short", "password2": "short"},
    {"username": "amy", "password": "abc12345", "password2": "abc99999"},
]:
    try:
        Register(**data)
    except ValidationError as e:
        print("❌", e.errors()[0]["msg"])
```

## 巢狀模型

真實的 API 資料常常是巢狀的：

```python
from pydantic import BaseModel, Field

class OrderItem(BaseModel):
    product: str
    price: int = Field(gt=0)
    qty: int = Field(ge=1)

    @property
    def subtotal(self) -> int:
        return self.price * self.qty

class Address(BaseModel):
    city: str
    street: str

class Order(BaseModel):
    customer: str
    items: list[OrderItem] = Field(min_length=1)
    shipping: Address

    @property
    def total(self) -> int:
        return sum(i.subtotal for i in self.items)

data = {
    "customer": "小明",
    "items": [{"product": "耳機", "price": 1290, "qty": 1}, {"product": "USB 線", "price": 290, "qty": "2"}],
    "shipping": {"city": "台北市", "street": "信義路五段 7 號"},
}
order = Order(**data)                 # 也可以用 Order.model_validate(data)
print(order.shipping.city, len(order.items), "件商品，總計", order.total, "元")
print(order.model_dump_json(indent=2))
```

## 讀取 JSON

```python
from pydantic import BaseModel

class Config(BaseModel):
    app_name: str
    debug: bool = False
    max_users: int = 100

raw = '{"app_name": "我的網站", "debug": "true", "max_users": "500"}'
cfg = Config.model_validate_json(raw)
print(cfg, type(cfg.debug), type(cfg.max_users))
```

## 在 FastAPI 裡

```py
from fastapi import FastAPI
from pydantic import BaseModel, Field

app = FastAPI()

class OrderIn(BaseModel):
    product_id: int
    qty: int = Field(ge=1, le=99)

class OrderOut(BaseModel):
    order_id: int
    product_id: int
    qty: int
    total: int

@app.post("/orders", response_model=OrderOut)     # response_model：也會檢查回傳的資料
def create_order(order: OrderIn):
    return OrderOut(order_id=1001, total=order.qty * 590, **order.model_dump())
```

`/docs` 頁面會自動顯示這些模型的格式和限制。

@@solution
from typing import Literal
from pydantic import BaseModel, Field, field_validator, ValidationError


class Registration(BaseModel):
    name: str = Field(min_length=1, max_length=20)
    age: int = Field(ge=12, le=99)
    email: str
    sessions: list[Literal["上午", "下午", "晚上"]] = Field(min_length=1)

    @field_validator("email")
    @classmethod
    def check_email(cls, v: str) -> str:
        if v.count("@") != 1 or "." not in v.split("@")[1]:
            raise ValueError("email 格式錯誤")
        return v.lower()


print(Registration(name="小明", age=18, email="Ming@Example.com", sessions=["上午", "晚上"]))
