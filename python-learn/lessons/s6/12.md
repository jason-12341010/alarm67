第 6 階段的期末專案：用 **FastAPI + SQLAlchemy + Pydantic** 做一個完整的**待辦事項 API**，包含測試和部署。這是後端工程師最經典的練習。

## 功能規劃

| 方法 | 路徑 | 功能 |
|---|---|---|
| `GET` | `/todos` | 列出待辦事項（可篩選：`?done=false`、`?q=關鍵字`） |
| `POST` | `/todos` | 新增 |
| `GET` | `/todos/{id}` | 查看一筆 |
| `PATCH` | `/todos/{id}` | 修改部分欄位（例如只標記完成） |
| `DELETE` | `/todos/{id}` | 刪除 |
| `GET` | `/stats` | 統計：總數、完成數、逾期數 |

## Part 1：在網頁上先做出核心邏輯

先不管 HTTP，把**資料模型、驗證、資料庫操作**做好。這部分在網頁上就能執行：

```python
from datetime import date
from pydantic import BaseModel, Field
from sqlalchemy import create_engine, select, func
from sqlalchemy.orm import DeclarativeBase, Mapped, mapped_column, Session


# ---------- 資料庫模型（SQLAlchemy） ----------
class Base(DeclarativeBase):
    pass


class TodoRow(Base):
    __tablename__ = "todos"
    id: Mapped[int] = mapped_column(primary_key=True)
    title: Mapped[str]
    done: Mapped[bool] = mapped_column(default=False)
    priority: Mapped[int] = mapped_column(default=2)
    due: Mapped[date | None] = mapped_column(default=None)


# ---------- API 的輸入輸出格式（Pydantic） ----------
class TodoCreate(BaseModel):
    title: str = Field(min_length=1, max_length=100)
    priority: int = Field(default=2, ge=1, le=3)       # 1 高、2 中、3 低
    due: date | None = None


class TodoUpdate(BaseModel):                           # PATCH：每個欄位都是選填
    title: str | None = Field(default=None, min_length=1, max_length=100)
    done: bool | None = None
    priority: int | None = Field(default=None, ge=1, le=3)
    due: date | None = None


class TodoOut(BaseModel):
    model_config = {"from_attributes": True}           # 可以直接從 SQLAlchemy 物件轉換
    id: int
    title: str
    done: bool
    priority: int
    due: date | None


# ---------- 資料存取層 ----------
class TodoRepo:
    def __init__(self, session: Session):
        self.s = session

    def create(self, data: TodoCreate) -> TodoOut:
        row = TodoRow(**data.model_dump())
        self.s.add(row)
        self.s.commit()
        return TodoOut.model_validate(row)

    def list(self, done: bool | None = None, q: str | None = None) -> list[TodoOut]:
        stmt = select(TodoRow).order_by(TodoRow.done, TodoRow.priority, TodoRow.id)
        if done is not None:
            stmt = stmt.where(TodoRow.done == done)
        if q:
            stmt = stmt.where(TodoRow.title.contains(q))
        return [TodoOut.model_validate(r) for r in self.s.scalars(stmt)]

    def update(self, todo_id: int, data: TodoUpdate) -> TodoOut | None:
        row = self.s.get(TodoRow, todo_id)
        if row is None:
            return None
        for key, value in data.model_dump(exclude_unset=True).items():   # 只更新有傳的欄位
            setattr(row, key, value)
        self.s.commit()
        return TodoOut.model_validate(row)

    def delete(self, todo_id: int) -> bool:
        row = self.s.get(TodoRow, todo_id)
        if row is None:
            return False
        self.s.delete(row)
        self.s.commit()
        return True

    def stats(self, today: date) -> dict:
        total = self.s.scalar(select(func.count()).select_from(TodoRow))
        done = self.s.scalar(select(func.count()).where(TodoRow.done))
        overdue = self.s.scalar(select(func.count()).where(~TodoRow.done, TodoRow.due < today))
        return {"total": total, "done": done, "overdue": overdue}


# ---------- 試試看 ----------
engine = create_engine("sqlite:///:memory:")
Base.metadata.create_all(engine)
with Session(engine) as session:
    repo = TodoRepo(session)
    repo.create(TodoCreate(title="寫報告", priority=1, due="2026-11-01"))
    repo.create(TodoCreate(title="買牛奶", priority=3))
    repo.create(TodoCreate(title="學 FastAPI", due="2026-11-30"))
    repo.update(2, TodoUpdate(done=True))
    for t in repo.list():
        print(("✅" if t.done else "⬜"), t.model_dump_json())
    print(repo.stats(today=date(2026, 11, 4)))
    print("搜尋「報告」：", [t.title for t in repo.list(q="報告")])
```

### `app/main.py`

```py
import os
from datetime import date
from fastapi import FastAPI, Depends, HTTPException
from sqlalchemy import create_engine
from sqlalchemy.orm import Session

from .models import Base, TodoCreate, TodoUpdate, TodoOut
from .repo import TodoRepo

engine = create_engine(os.environ.get("DATABASE_URL", "sqlite:///todo.db"))
Base.metadata.create_all(engine)
app = FastAPI(title="待辦事項 API", version="1.0")


def get_repo():
    with Session(engine) as session:
        yield TodoRepo(session)


@app.get("/todos", response_model=list[TodoOut])
def list_todos(done: bool | None = None, q: str | None = None, repo: TodoRepo = Depends(get_repo)):
    return repo.list(done=done, q=q)


@app.post("/todos", response_model=TodoOut, status_code=201)
def create_todo(data: TodoCreate, repo: TodoRepo = Depends(get_repo)):
    return repo.create(data)


@app.get("/todos/{todo_id}", response_model=TodoOut)
def get_todo(todo_id: int, repo: TodoRepo = Depends(get_repo)):
    todo = next((t for t in repo.list() if t.id == todo_id), None)
    if todo is None:
        raise HTTPException(404, "找不到待辦事項")
    return todo


@app.patch("/todos/{todo_id}", response_model=TodoOut)
def update_todo(todo_id: int, data: TodoUpdate, repo: TodoRepo = Depends(get_repo)):
    todo = repo.update(todo_id, data)
    if todo is None:
        raise HTTPException(404, "找不到待辦事項")
    return todo


@app.delete("/todos/{todo_id}", status_code=204)
def delete_todo(todo_id: int, repo: TodoRepo = Depends(get_repo)):
    if not repo.delete(todo_id):
        raise HTTPException(404, "找不到待辦事項")


@app.get("/stats")
def stats(repo: TodoRepo = Depends(get_repo)):
    return repo.stats(today=date.today())
```

os.environ["DATABASE_URL"] = "sqlite:///test.db"

from fastapi.testclient import TestClient

from app.models import Base

client = TestClient(app)

def test_create_and_list():
    r = client.post("/todos", json={"title": "寫測試", "priority": 1})
    assert r.status_code == 201
    assert r.json()["done"] is False
    assert len(client.get("/todos").json()) == 1


def test_patch_done():
    tid = client.post("/todos", json={"title": "a"}).json()["id"]
    r = client.patch(f"/todos/{tid}", json={"done": True})
    assert r.json()["done"] is True and r.json()["title"] == "a"     # 其他欄位不變


def test_validation_and_404():
    assert client.post("/todos", json={"title": ""}).status_code == 422

from datetime import date, timedelta
from pydantic import BaseModel


class TodoOut(BaseModel):
    id: int
    title: str
    done: bool = False
    priority: int = 2
    due: date | None = None


def due_soon(todos, today, days=3):
    end = today + timedelta(days=days)
    soon = [t for t in todos if not t.done and t.due is not None and today <= t.due <= end]
    return [t.title for t in sorted(soon, key=lambda t: (t.due, t.priority))]


todos = [
    TodoOut(id=1, title="繳電費", due="2026-11-05", priority=1),
    TodoOut(id=2, title="交報告", due="2026-11-07"),
    TodoOut(id=3, title="看牙醫", due="2026-11-05", priority=2),
    TodoOut(id=4, title="買禮物", due="2026-11-20"),
    TodoOut(id=5, title="已完成的", due="2026-11-04", done=True),

]
