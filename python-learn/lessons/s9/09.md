## Part 2：從零打造的神經網路

把前面幾課學的全部組合起來，寫成一個可重複使用的類別：

```python
import numpy as np
from sklearn.datasets import load_digits
from sklearn.model_selection import train_test_split

class NumpyMLP:
    def __init__(self, sizes, seed=0):
        rng = np.random.default_rng(seed)
        # He 初始化：讓 ReLU 網路一開始的數值不會太大或太小
        self.W = [rng.normal(0, np.sqrt(2 / a), (a, b)) for a, b in zip(sizes[:-1], sizes[1:])]
        self.b = [np.zeros(b) for b in sizes[1:]]
        self.rng = rng

    @staticmethod
    def softmax(z):
        e = np.exp(z - z.max(axis=1, keepdims=True))
        return e / e.sum(axis=1, keepdims=True)

    def forward(self, X):
        self.acts = [X]
        for i, (W, b) in enumerate(zip(self.W, self.b)):
            z = self.acts[-1] @ W + b
            self.acts.append(self.softmax(z) if i == len(self.W) - 1 else np.maximum(0, z))
        return self.acts[-1]

    def backward(self, Y, lr):
        grad = (self.acts[-1] - Y) / len(Y)                 # softmax + 交叉熵
        for i in reversed(range(len(self.W))):
            dW, db = self.acts[i].T @ grad, grad.sum(axis=0)
            if i > 0:
                grad = (grad @ self.W[i].T) * (self.acts[i] > 0)   # ReLU 的梯度
            self.W[i] -= lr * dW
            self.b[i] -= lr * db

    def fit(self, X, y, epochs=30, lr=0.1, batch_size=32, verbose=True):
        Y = np.eye(self.W[-1].shape[1])[y]
        for epoch in range(epochs):
            idx = self.rng.permutation(len(X))
            for s in range(0, len(X), batch_size):
                b = idx[s:s + batch_size]
                self.forward(X[b])
                self.backward(Y[b], lr)
            if verbose and epoch % 5 == 0:
                p = self.forward(X)
                loss = -np.log(p[np.arange(len(y)), y] + 1e-9).mean()
                print(f"epoch {epoch:>2}  loss {loss:.4f}  train acc {(p.argmax(1) == y).mean():.3f}")
        return self

    def predict(self, X):
        return self.forward(X).argmax(axis=1)

digits = load_digits()
X = digits.data / 16.0                                     # 縮放到 0～1
X_train, X_test, y_train, y_test = train_test_split(X, digits.target, test_size=0.2, random_state=0, stratify=digits.target)

net = NumpyMLP([64, 64, 10]).fit(X_train, y_train, epochs=30)
print(f"\n✅ 測試準確率：{(net.predict(X_test) == y_test).mean():.1%}")
```

    ax.imshow(X_test[i].reshape(8, 8), cmap="gray_r")
    ax.set_title(f"答 {y_test[i]}\n猜 {pred[i]}", color="red", fontsize=10)

## Part 5：PyTorch + CNN 挑戰 MNIST（在 Colab 執行）

把下面的程式**整段**貼到 Colab（記得開 GPU），大約 2 分鐘就能訓練完：

```py
import torch
from torch import nn
from torch.utils.data import DataLoader
from torchvision import datasets, transforms

device = "cuda" if torch.cuda.is_available() else "cpu"
tf = transforms.Compose([transforms.ToTensor(), transforms.Normalize((0.1307,), (0.3081,))])
train_ds = datasets.MNIST("data", train=True, download=True, transform=tf)
test_ds = datasets.MNIST("data", train=False, download=True, transform=tf)
train_loader = DataLoader(train_ds, batch_size=128, shuffle=True)
test_loader = DataLoader(test_ds, batch_size=1000)

model = nn.Sequential(
    nn.Conv2d(1, 32, 3, padding=1), nn.ReLU(), nn.MaxPool2d(2),
    nn.Conv2d(32, 64, 3, padding=1), nn.ReLU(), nn.MaxPool2d(2),
    nn.Flatten(),
    nn.Linear(64 * 7 * 7, 128), nn.ReLU(), nn.Dropout(0.3),
    nn.Linear(128, 10),
).to(device)
optimizer = torch.optim.Adam(model.parameters(), lr=1e-3)
loss_fn = nn.CrossEntropyLoss()

for epoch in range(5):
    model.train()
    for xb, yb in train_loader:
        xb, yb = xb.to(device), yb.to(device)
        loss = loss_fn(model(xb), yb)
        optimizer.zero_grad()
        loss.backward()
        optimizer.step()

    model.eval()
    correct = 0
    with torch.no_grad():
        for xb, yb in test_loader:
            correct += (model(xb.to(device)).argmax(1) == yb.to(device)).sum().item()
    print(f"epoch {epoch + 1}：測試準確率 {correct / len(test_ds):.2%}")

torch.save(model.state_dict(), "mnist_cnn.pt")
```

預期結果：第 5 個 epoch 的準確率大約 **99.0%～99.3%**。
