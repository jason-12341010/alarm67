## 定義模型：nn.Module

```py
import torch
from torch import nn

class MLP(nn.Module):
    def __init__(self, n_in, n_hidden, n_out):
        super().__init__()
        self.net = nn.Sequential(
            nn.Linear(n_in, n_hidden),     # 全連接層：y = xW + b
            nn.ReLU(),
            nn.Dropout(0.2),               # 訓練時隨機關掉 20% 的神經元，防止過擬合
            nn.Linear(n_hidden, n_hidden),
            nn.ReLU(),
            nn.Linear(n_hidden, n_out),
        )

    def forward(self, x):                  # 定義前向傳播
        return self.net(x)

model = MLP(4, 32, 3)
print(model)
print("參數數量：", sum(p.numel() for p in model.parameters()))
out = model(torch.randn(5, 4))            # 呼叫 model(x) 會自動執行 forward
print(out.shape)                           # torch.Size([5, 3])
```

## 資料：Dataset 和 DataLoader

`DataLoader` 負責把資料**分批（batch）**、**打亂順序**：

```py
import torch
from torch.utils.data import TensorDataset, DataLoader
from sklearn.datasets import load_iris
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler

X, y = load_iris(return_X_y=True)
X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.2, random_state=0, stratify=y)
scaler = StandardScaler().fit(X_train)
X_train, X_test = scaler.transform(X_train), scaler.transform(X_test)

train_ds = TensorDataset(torch.tensor(X_train, dtype=torch.float32), torch.tensor(y_train))
test_ds = TensorDataset(torch.tensor(X_test, dtype=torch.float32), torch.tensor(y_test))
train_loader = DataLoader(train_ds, batch_size=16, shuffle=True)
test_loader = DataLoader(test_ds, batch_size=64)

xb, yb = next(iter(train_loader))
print(xb.shape, yb.shape)       # torch.Size([16, 4]) torch.Size([16])
```

## 完整的訓練迴圈

```py
import torch
from torch import nn

device = "cuda" if torch.cuda.is_available() else "cpu"
model = MLP(4, 32, 3).to(device)
loss_fn = nn.CrossEntropyLoss()                                # 多類別分類的損失（內含 softmax）
optimizer = torch.optim.Adam(model.parameters(), lr=1e-3)      # Adam：最常用的優化器

for epoch in range(100):
    model.train()                                  # 訓練模式（啟用 Dropout）
    total_loss = 0
    for xb, yb in train_loader:
        xb, yb = xb.to(device), yb.to(device)
        pred = model(xb)                           # 1. 前向傳播
        loss = loss_fn(pred, yb)                   # 2. 計算損失
        optimizer.zero_grad()                      # 3. 梯度歸零
        loss.backward()                            # 4. 反向傳播
        optimizer.step()                           # 5. 更新權重
        total_loss += loss.item() * len(xb)

    model.eval()                                   # 評估模式（關閉 Dropout）
    correct = 0
    with torch.no_grad():                          # 不需要計算梯度
        for xb, yb in test_loader:
            pred = model(xb.to(device)).argmax(dim=1)
            correct += (pred == yb.to(device)).sum().item()
    if epoch % 20 == 0:
        print(f"epoch {epoch:>3}  loss {total_loss / len(train_ds):.4f}  test acc {correct / len(test_ds):.3f}")

torch.save(model.state_dict(), "iris_mlp.pt")      # 存檔
```

## 在網頁上：用 NumPy 實作小批次訓練

小批次（mini-batch）訓練的概念完全一樣，我們用 NumPy 把它寫出來：

```python
import numpy as np
from sklearn.datasets import load_iris
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler

X, y = load_iris(return_X_y=True)
X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.2, random_state=0, stratify=y)
scaler = StandardScaler().fit(X_train)
X_train, X_test = scaler.transform(X_train), scaler.transform(X_test)
Y_train = np.eye(3)[y_train]                         # One-Hot 編碼

rng = np.random.default_rng(0)
W1, b1 = rng.normal(0, 0.5, (4, 16)), np.zeros(16)
W2, b2 = rng.normal(0, 0.5, (16, 3)), np.zeros(3)

def softmax(z):
    e = np.exp(z - z.max(axis=1, keepdims=True))
    return e / e.sum(axis=1, keepdims=True)

def predict(X):
    return softmax(np.maximum(0, X @ W1 + b1) @ W2 + b2)

lr, batch_size = 0.1, 16
for epoch in range(60):
    idx = rng.permutation(len(X_train))               # 每個 epoch 打亂順序
    for start in range(0, len(idx), batch_size):      # 一次取一小批
        b = idx[start:start + batch_size]
        xb, yb = X_train[b], Y_train[b]
        h = np.maximum(0, xb @ W1 + b1)
        p = softmax(h @ W2 + b2)
        dz2 = (p - yb) / len(b)                       # softmax + 交叉熵的梯度
        dW2, db2 = h.T @ dz2, dz2.sum(0)
        dz1 = (dz2 @ W2.T) * (h > 0)                  # ReLU 的梯度
        dW1, db1 = xb.T @ dz1, dz1.sum(0)
        W1 -= lr * dW1; b1 -= lr * db1; W2 -= lr * dW2; b2 -= lr * db2
    if epoch % 10 == 0:
        p = predict(X_train)
        loss = -np.log(p[np.arange(len(y_train)), y_train]).mean()
        acc = (predict(X_test).argmax(1) == y_test).mean()
        print(f"epoch {epoch:>2}  loss {loss:.4f}  test acc {acc:.3f}")
```

## 常見的訓練技巧

| 技巧 | 用途 |
|---|---|
| **Dropout** | 隨機關閉神經元，防止過擬合 |
| **Batch Normalization** | 讓每一層的輸入保持穩定，訓練更快 |
| **Early Stopping** | 驗證集的損失不再下降就停止訓練 |
| **學習率排程** | 一開始學快一點，後面慢慢變小 |
| **權重衰減** | 懲罰太大的權重（就是 L2 正則化） |
| **資料增強** | 圖片隨機翻轉、裁切，增加資料的多樣性 |

def softmax(z):

    return e / e.sum(axis=1, keepdims=True)


def cross_entropy(p, y):
    return -np.log(p[np.arange(len(y)), y]).mean()
